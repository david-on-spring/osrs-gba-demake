  if near WHEAT_FIELD 4 goto wheat
  if near ONION_FIELD 4 goto onion
  if near CABBAGE_FIELD 5 goto cabbage
  chat "These crops aren't ready to be picked yet."
  end
:wheat
  if not free 1 goto full
  give GRAIN
  sfx PICKUP
  chat "You pick some wheat."
  end
:onion
  if not free 1 goto full
  give ONION
  sfx PICKUP
  chat "You pick an onion."
  end
:cabbage
  if not free 1 goto full
  give CABBAGE
  sfx PICKUP
  chat "You pick a cabbage."
  end
:full
  chat "You don't have enough inventory space."

obj WATER
  if not has BUCKET goto none
  take BUCKET
  give BUCKET_WATER
  sfx SPLASH
  chat "You fill the bucket with water."
:none

obj FOUNTAIN
  if near MANOR_FOUNTAIN 2 goto manor
:fill
  if not has BUCKET goto none
  take BUCKET
  give BUCKET_WATER
  sfx SPLASH
  chat "You fill the bucket from the fountain."
  end
:none
  chat "The water looks clean and cool."
  end
:manor
  if flag ERNEST 0 goto fill
  if has GAUGE goto fill
  if flag ERNEST 4 goto search
  if has POISON_FOOD goto poison
  narrate "There's a pressure gauge at the bottom of the fountain, but the water is full of very hungry-looking piranhas. I'm not putting my hand in there."
  end
:poison
  take POISON_FOOD
  flag ERNEST 4
  narrate "You sprinkle the poisoned fish food into the fountain. The piranhas gobble it up... and soon float belly-up."
:search
  give GAUGE
  sfx PICKUP
  narrate "You search the fountain and find the pressure gauge."

obj ALTAR
  if near TOWER_ALTAR 1 goto tower
  chat "You kneel and pray at the altar. You feel at peace."
  end
:tower
  if stage GHOST == 3 goto take
  if stage GHOST == 4 goto again
  chat "An old altar. Something used to rest on it."
  end
:again
  if has SKULL goto taken
:take
  give SKULL
  stage GHOST 4
  sfx DEATH
  narrate "You take the skull from the altar. With a clatter of bones, a skeleton rises to guard it!"
  end
:taken
  chat "There's nothing else on the altar."

# ============================================================================================
# Cook's Assistant
# ============================================================================================
talk COOK
  if done COOK goto finished
  if started COOK goto progress
  npc "What am I to do?"
  choice
    option "What's wrong?" wrong
    option "Can you make me a cake?" wrong
    option "You don't look very happy." wrong
    option "Nice hat!" hat
:hat
  player "Nice hat!"
  npc "Er, thank you. It's a pretty ordinary cook's hat really."
  player "Still, it suits you. The trousers don't, though."
  npc "Hey! These are standard issue, I'll have you know. Anyway, I've got bigger problems than my trousers."
:wrong
  npc "Oh dear, oh dear, oh dear, I'm in a terrible, terrible mess! It's the Duke's birthday today, and I should be making him a lovely big birthday cake."
  npc "But I've forgotten to buy the ingredients! I'll never get them in time now. He'll sack me! What will I do? I have four children and a goat to look after."
  npc "Would you help me? Please?"
  choice
    option "I'm always happy to help a cook in distress." yes
    option "I can't right now, maybe later." no
:no
  player "I can't right now, maybe later."
  npc "Fine. I always knew you adventurer types were callous beasts. Go on your merry way!"
  end
:yes
  player "Yes, I'll help you."
  npc "Oh thank you, thank you! I need milk, an egg and flour. I'd be very grateful if you could get them for me."
  stage COOK 1
  player "So where do I find these ingredients then?"
  npc "There's a mill north-west of here, Mill Lane Mill. Pick some wheat from the field beside it, put the grain in the hopper and collect the flour from the bin with an empty pot. There's a pot on my table."
  npc "The dairy cow is in the cow field east of the farms; you'll need a bucket to milk her. And the chickens north of here lay their eggs in the coops."
  npc "Hurry, please!"
  end
:progress
  npc "How are you getting on with finding the ingredients?"
  if flag COOK 0 goto c_milk
  if not has EGG goto c_milk
  take EGG
  flag COOK 0
  player "Here's a fresh egg."
:c_milk
  if flag COOK 1 goto c_flour
  if not has BUCKET_MILK goto c_flour
  take BUCKET_MILK
  give BUCKET
  flag COOK 1
  player "Here's a bucket of milk."
:c_flour
  if flag COOK 2 goto check
  if not has POT_FLOUR goto check
  take POT_FLOUR
  give POT
  flag COOK 2
  player "Here's a pot of flour."
:check
  if not flag COOK 0 goto notyet
  if not flag COOK 1 goto notyet
  if not flag COOK 2 goto notyet
  npc "You've brought me everything I need! I am saved! Thank you!"
  player "So do I get to go to the Duke's party?"
  npc "I'm afraid not, only the big cheeses get to dine with the Duke."
  player "Well, maybe one day I'll be important enough to sit at the Duke's table."
  npc "Maybe, but I won't be holding my breath."
  complete COOK
  end
:notyet
  npc "I still need:"
  if flag COOK 0 goto n_milk
  npc "A fresh egg - try the chicken coops north of the castle."
:n_milk
  if flag COOK 1 goto n_flour
  npc "A bucket of milk - the dairy cow is in the cow field east of the farms."
:n_flour
  if flag COOK 2 goto n_end
  npc "A pot of flour - grind wheat at Mill Lane Mill and fill a pot from the flour bin."
:n_end
  end
:finished
  npc "Hello, friend! How is the adventuring going?"
  choice
    option "I'm getting strong and mighty." strong
    option "I keep on dying." dying
    option "Can I use your range?" range
:strong
  npc "Glad to hear it."
  end
:dying
  npc "Ah well, at least you keep coming back to life!"
  end
:range
  npc "Go ahead! It's a very good range. It's easier to use than most, so you'll burn less food on it."

talk DAIRY_COW
  if not has BUCKET goto none
  take BUCKET
  give BUCKET_MILK
  sfx SPLASH
  chat "You milk the cow."
  end
:none
  chat "You'll need an empty bucket to collect the milk."

obj HOPPER
  if flag COOK 3 goto full
  if not has GRAIN goto none
  take GRAIN
  flag COOK 3
  sfx MINE
  chat "You put the grain in the hopper and operate the controls."
  chat "The grain slides down the chute into the flour bin."
  end
:full
  chat "There is already grain in the hopper. Fetch the flour from the bin."
  end
:none
  chat "The hopper is empty. It needs grain from the wheat field beside the mill."

obj FLOURBIN
  if not flag COOK 3 goto empty
  if not has POT goto nopot
  take POT
  give POT_FLOUR
  unflag COOK 3
  sfx PICKUP
  chat "You fill a pot with flour from the bin."
  end
:nopot
  chat "There is flour in the bin. You need an empty pot to hold it."
  end
:empty
  chat "The flour bin is empty."

# ============================================================================================
# Sheep Shearer
# ============================================================================================
talk FRED
  if done SHEEP goto finished
  if started SHEEP goto progress
  npc "What are you doing on my land? You're not the one who keeps leaving all my gates open and letting out all my sheep, are you?"
  choice
    option "I'm looking for a quest." quest
    option "I'm looking for something to kill." kill
    option "I'm lost." lost
:kill
  npc "What, on my land? Leave my livestock alone, you scoundrel!"
  end
:lost
  npc "How can you be lost? Just follow the road east and south. You'll end up in Lumbridge fairly quickly."
  end
:quest
  player "I'm looking for a quest."
  npc "You're after a quest, you say? Actually I could do with a bit of help."
  npc "My sheep are getting mighty woolly. I'd be much obliged if you could shear them. And while you're at it, spin the wool for me too."
  npc "Bring me twenty balls of wool and I'll pay you for your trouble."
  choice
    option "Yes okay. I can do that." yes
    option "That doesn't sound a very exciting quest." boring
:boring
  npc "Well what do you expect if you ask a farmer for a quest?"
  end
:yes
  player "Yes okay. I can do that."
  stage SHEEP 1
  npc "Good! Now one more thing, do you actually know how to shear a sheep?"
  choice
    option "Of course!" know
    option "Err. No, I don't know actually." dontknow
:dontknow
  npc "Well, first things first, you need a pair of shears. There's a pair in my house, or the Lumbridge general store sells them."
  npc "Walk up to a sheep and shear it. Then take the wool to the spinning wheel in Lumbridge Castle and spin it into balls."
:know
  npc "Off you go then! The sheep are in the field south-east of my house."
  end
:progress
  npc "How are you doing getting those balls of wool?"
  if not has BALL_WOOL goto none
  player "I have some."
  npc "Give 'em here then."
:give
  if var SHEEP >= 20 goto finish
  if not has BALL_WOOL goto more
  take BALL_WOOL
  addvar SHEEP 1
  goto give
:more
  npc "That's not all of them. I need twenty balls of wool in all - check your quest journal to see how many you've brought me."
  end
:none
  player "I haven't got any at the moment."
  npc "Ah well, at least you haven't been eaten. Shear the sheep, spin the wool, and bring me twenty balls."
  end
:finish
  player "That's the last of them."
  npc "I guess I'd better pay you then. Thanks for your help!"
  complete SHEEP
  end
:finished
  npc "Thanks for shearing my sheep. I couldn't have done it without you."

talk SHEEP
  if npcstate goto shorn
  if not has SHEARS goto noshears
  if not free 1 goto full
  give WOOL
  npcstate 60
  sfx PICKUP
  chat "You get some wool."
  end
:shorn
  chat "This sheep has already been shorn."
  end
:noshears
  chat "You need a set of shears to do this."
  end
:full
  chat "You don't have enough inventory space."

obj WHEEL
  if not has WOOL goto nowool
:spin
  if not has WOOL goto spun
  take WOOL
  give BALL_WOOL
  goto spin
:spun
  sfx PICKUP
  chat "You spin the wool into balls of wool."
  end
:nowool
  chat "You need some wool to spin."

talk AERECK
  if done GHOST goto finished
  if started GHOST goto progress
  npc "Welcome to the church of holy Saradomin."
  choice
    option "Who's Saradomin?" who
    option "Nice place you've got here." nice
    option "I'm looking for a quest." quest
:who
  npc "Surely you have heard of the god Saradomin? He who creates the forces of goodness and purity in this world? I cannot believe your ignorance!"
  end
:nice
  npc "It is, isn't it? It was built over two centuries ago."
  end
:quest
  player "I'm looking for a quest."
  npc "That's lucky, I need someone to do a quest for me."
  choice
    option "Ok, let me help then." yes
    option "Sorry, I don't have time right now." no
:no
  npc "Oh well. If you do have some spare time on your hands, come back and talk to me."
  end
:yes
  player "Ok, let me help then."
  npc "Thank you. The problem is, there is a ghost in the church graveyard. I would like you to get rid of it."
  npc "If you need any help, my friend Father Urhney is an expert on ghosts. He lives in a little shack in the south of Lumbridge Swamp."
  stage GHOST 1
  end
:progress
  npc "Have you got rid of the ghost yet?"
  if stage GHOST >= 3 goto p3
  if stage GHOST == 2 goto p2
  player "I can't find Father Urhney at the moment."
  npc "Well, you can get to the swamp he lives in by going south-west, past the graveyard."
  end
:p2
  player "I had a talk with Father Urhney. He has given me this funny amulet to talk to the ghost with."
  npc "I always wondered what that amulet was... Well, I hope it's useful. Tell me when you get rid of the ghost!"
  end
:p3
  player "The ghost wants his skull back. He thinks a warlock left it in the Wizards' Tower."
  npc "Then I hope you find it soon. Saradomin be with you."
  end
:finished
  npc "Thank you for getting rid of that awful ghost for me! May Saradomin always smile upon you!"

talk URHNEY
  npc "Go away! I'm meditating!"
  if stage GHOST == 1 goto help
  if stage GHOST >= 2 goto lost
  choice
    option "Well, that's friendly." friendly
    option "I've come to repossess your house." house
:friendly
  npc "I said go away!"
  end
:house
  npc "Under what grounds? ...Actually, I don't care. Go away."
  end
:help
  choice
    option "Father Aereck sent me to talk to you." aereck
    option "Well, that's friendly." friendly
:aereck
  player "Father Aereck sent me to talk to you."
  npc "I suppose I'd better talk to you then. What problems has he got himself into this time?"
  player "He's got a ghost haunting his graveyard."
  npc "Oh, the silly fool. I leave town for just five months, and already he can't manage. Well, I can't go back and exorcise it. I vowed not to leave this place."
  npc "Tell you what I can do though: take this amulet. It is an Amulet of Ghostspeak. Wear it and you'll be able to talk to the ghost and find out what's keeping it from resting."
  give GHOSTSPEAK
  stage GHOST 2
  chatc GOLD "Father Urhney hands you an amulet. Wear it from the Equipment menu."
  end
:lost
  if done GHOST goto bye
  if has GHOSTSPEAK goto bye
  player "I've lost the Amulet of Ghostspeak."
  npc "How careless can you get? Those things aren't easy to come by, you know! It's a good job I've got a spare."
  give GHOSTSPEAK
  chat "Father Urhney hands you an amulet."
  end
:bye
  npc "Now leave me in peace."

obj COFFIN
  if done GHOST goto rest
  if stage GHOST == 4 goto skull
  if not stage GHOST >= 2 goto shut
  if flag GHOST 0 goto open
  flag GHOST 0
  sfx DEATH
  narrate "You open the coffin. Inside lies a skeleton without a skull... and a ghost rises from it!"
  end
:open
  chat "The coffin is open. The skeleton inside is missing its skull."
  end
:shut
  chat "A coffin. It's best not to disturb the dead without a reason."
  end
:rest
  chat "The skeleton in the coffin rests peacefully now."
  end
:skull
  if not has SKULL goto noskull
  take SKULL
  narrate "You put the skull back in the coffin. The ghost sighs with relief and slowly fades away. At last he can rest."
  complete GHOST
  end
:noskull
  chat "The ghost's skull belongs in here."

talk GHOST
  if worn GHOSTSPEAK goto speak
  npc "Wooo wooo wooooo!"
  narrate "The ghost wails, but you can't understand a word. Perhaps wearing an Amulet of Ghostspeak would help."
  end
:speak
  if stage GHOST >= 3 goto skullq
  player "Hello ghost, how are you?"
  npc "Not very good actually."
  player "What's the problem then?"
  npc "Did you just understand what I said???"
  choice
    option "Yep, now tell me what the problem is." problem
    option "No, you sound like you're speaking nonsense." nonsense
:nonsense
  npc "Oh that's a pity. You got my hopes up there."
  player "Yeah, it is a pity. Sorry about that."
  npc "Hang on a second... you CAN understand me!"
:problem
  npc "WOW! This is INCREDIBLE! I didn't expect anyone to ever understand me again!"
  player "Ok, ok, I can understand you! But have you any idea WHY you're doomed to be a ghost?"
  npc "Well, to be honest... I think it's because a warlock stole my skull. I can't rest without it."
  player "Do you know where it is?"
  npc "I think he took it to the Wizards' Tower, south of Draynor, and left it on the altar there. Please, bring it back!"
  player "Ok. I will get the skull back for you, then you can rest in peace."
  stage GHOST 3
  end
:skullq
  if has SKULL goto haveskull
  npc "Have you found my skull yet? It should be on the altar in the Wizards' Tower."
  end
:haveskull
  npc "My skull! Please, put it back in my coffin so I can rest."
